// Edge Function : initiate-payin // Rôle : quand l'acheteur clique "Payer", cette fonction demande à Lemonway une page de // paiement hébergée (WebKit) pour le montant total (prix marteau + frais), et renvoie son URL. // Le navigateur redirige l'acheteur dessus. Le vrai encaissement est confirmé plus tard // par le webhook (lemonway-webhook), jamais par ce fichier ni par le client. // // Structure alignée sur la doc réelle de Lemonway (POST /moneyins/card/webinit) au 28/08/2026 : // https://documentation.lemonway.com/reference/moneyins_cardwebinitpost // // ⚠️ À COMPLÉTER une fois le compte Lemonway actif (voir prérequis ci-dessous) : // Secrets à définir dans Supabase (Project Settings > Edge Functions > Secrets) : // LEMONWAY_BEARER_TOKEN — jeton OAuth2 (valable 90 jours, généré depuis leur Dashboard : // Developers > API Keys, puis échangé contre un Bearer token — // voir https://documentation.lemonway.com/reference/authentication-oauth). // Pas de génération automatique ici : à renouveler manuellement tous les ~90 jours. // LEMONWAY_API_BASE — ex. https://sandbox-api.lemonway.fr/mb//directkitrest/v2 // (le segment est propre à ton compte, communiqué à l'onboarding) // LEMONWAY_WEBKIT_BASE — ex. https://sandbox-webkit.lemonway.fr//dev/ // (fourni dans l'email d'onboarding Lemonway) // LEMONWAY_WALLET_ID — l'accountId du wallet Kriee qui reçoit l'argent (accountId ci-dessous) import { createClient } from 'jsr:@supabase/supabase-js@2'; Deno.serve(async (req) => { try { const { sale_id } = await req.json(); const supabase = createClient( Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')! ); const authHeader = req.headers.get('Authorization'); const { data: { user } } = await supabase.auth.getUser(authHeader?.replace('Bearer ', '')); if (!user) return new Response(JSON.stringify({ error: 'Non authentifié.' }), { status: 401 }); const { data: sale, error } = await supabase .from('sales') .select('*') .eq('id', sale_id) .eq('buyer_id', user.id) .single(); if (error || !sale) { return new Response(JSON.stringify({ error: "Vente introuvable ou tu n'es pas l'acheteur." }), { status: 404 }); } if (sale.seller_accepted !== true) { return new Response(JSON.stringify({ error: 'Le vendeur doit accepter la vente avant tout paiement.' }), { status: 400 }); } const bearerToken = Deno.env.get('LEMONWAY_BEARER_TOKEN'); const apiBase = Deno.env.get('LEMONWAY_API_BASE'); const webkitBase = Deno.env.get('LEMONWAY_WEBKIT_BASE'); const walletId = Deno.env.get('LEMONWAY_WALLET_ID'); if (!bearerToken || !apiBase || !webkitBase || !walletId) { return new Response(JSON.stringify({ error: "Paiement pas encore activé : secrets Lemonway manquants côté serveur (voir les TODO dans ce fichier).", }), { status: 501, headers: { 'Content-Type': 'application/json' } }); } // Montants en centimes, exigé par l'API Lemonway (totalAmount, commissionAmount) const totalAmountCents = Math.round(sale.buyer_total_price * 100); const commissionAmountCents = Math.round((sale.buyer_total_price - sale.hammer_price) * 100); const clientIp = req.headers.get('x-forwarded-for')?.split(',')[0]?.trim() || '0.0.0.0'; const lemonwayRes = await fetch(`${apiBase}/moneyins/card/webinit`, { method: 'POST', headers: { 'Authorization': `Bearer ${bearerToken}`, 'Content-Type': 'application/json', 'PSU-IP-Address': clientIp, // obligatoire côté Lemonway }, body: JSON.stringify({ accountId: walletId, totalAmount: totalAmountCents, commissionAmount: commissionAmountCents, autoCommission: false, // sinon Lemonway remplace notre commission par la sienne reference: sale_id, // permet de relier le paiement à la vente dans le webhook label: 'Kriee — enchère nautique', returnUrl: `https://kriee.vercel.app/compte.html?paiement=ok&sale=${sale_id}#activite`, cancelUrl: `https://kriee.vercel.app/compte.html?paiement=annule&sale=${sale_id}#activite`, errorUrl: `https://kriee.vercel.app/compte.html?paiement=erreur&sale=${sale_id}#activite`, }), }); if (!lemonwayRes.ok) { const errText = await lemonwayRes.text(); return new Response(JSON.stringify({ error: `Erreur Lemonway : ${errText}` }), { status: 502 }); } const lemonwayData = await lemonwayRes.json(); // La doc indique un MoneyInToken en retour ; le nom exact du champ JSON n'est pas confirmé // dans la doc textuelle consultée — à vérifier sur un vrai appel sandbox avant mise en prod. const token = lemonwayData.token || lemonwayData.moneyInToken || lemonwayData.MoneyInToken; await supabase.from('sales').update({ payment_status: 'pending' }).eq('id', sale_id); return new Response(JSON.stringify({ payment_url: `${webkitBase}${token}`, }), { headers: { 'Content-Type': 'application/json' } }); } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 500 }); } });
